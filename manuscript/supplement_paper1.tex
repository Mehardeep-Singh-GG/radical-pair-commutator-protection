\subsection{Electronic Singlet--Triplet Algebra and Symmetrized Dipolar Operators}
\label{subsec:S1_elec}

Let $\mathcal{H}_e := \mathbb{C}^2 \otimes \mathbb{C}^2$ be the four-dimensional two-electron spin space with Pauli matrices $\sigma_1 = \sigma_x, \sigma_2 = \sigma_y, \sigma_3 = \sigma_z$ and spin-$1/2$ operators $S_{1\alpha} = \frac{1}{2}\sigma_\alpha \otimes \mathbb{I}_2$, $S_{2\alpha} = \frac{1}{2}\mathbb{I}_2 \otimes \sigma_\alpha$ ($\alpha \in \{1,2,3\}$). They satisfy the canonical $\mathfrak{su}(2)$ relations
\begin{equation}
\label{eq:S1_su2}
[S_{e\alpha}, S_{e'\beta}] = \ii\,\delta_{ee'}\sum_{\gamma=1}^3 \epsilon_{\alpha\beta\gamma} S_{e\gamma}, \qquad
S_{e\alpha} S_{e\beta} = \frac{1}{4}\delta_{\alpha\beta}\mathbb{I}_4 + \frac{\ii}{2}\sum_{\gamma=1}^3 \epsilon_{\alpha\beta\gamma} S_{e\gamma},
\end{equation}
for $e, e' \in \{1,2\}$. Define the singlet and triplet projectors on $\mathcal{H}_e$:
\begin{equation}
\label{eq:S1_proj}
P_{S,4} := \frac{1}{4}\mathbb{I}_4 - \mathbf{S}_1 \cdot \mathbf{S}_2, \qquad
P_{T,4} := \frac{3}{4}\mathbb{I}_4 + \mathbf{S}_1 \cdot \mathbf{S}_2 = \mathbb{I}_4 - P_{S,4}.
\end{equation}
In the orthonormal singlet--triplet basis
\begin{equation}
\label{eq:S1_st_basis}
|S\rangle = \frac{|\uparrow\downarrow\rangle - |\downarrow\uparrow\rangle}{\sqrt{2}}, \quad
|T_+\rangle = |\uparrow\uparrow\rangle, \quad
|T_0\rangle = \frac{|\uparrow\downarrow\rangle + |\downarrow\uparrow\rangle}{\sqrt{2}}, \quad
|T_-\rangle = |\downarrow\downarrow\rangle,
\end{equation}
we have $\mathbf{S}_1 \cdot \mathbf{S}_2 |S\rangle = -\frac{3}{4}|S\rangle$ and $\mathbf{S}_1 \cdot \mathbf{S}_2 |T_m\rangle = +\frac{1}{4}|T_m\rangle$ ($m \in \{+,0,-\}$). Thus $P_{S,4} = |S\rangle\langle S|$ and $P_{T,4} = \sum_m |T_m\rangle\langle T_m|$ immediately satisfy $P_{S,4}^2 = P_{S,4}$, $P_{T,4}^2 = P_{T,4}$, $P_{S,4}P_{T,4} = 0$, $P_{S,4}^\dagger = P_{S,4}$, $\Tr_e(P_{S,4}) = 1$, and $\Tr_e(P_{T,4}) = 3$.

Let $Z_\alpha := S_{1\alpha} + S_{2\alpha}$ be the total electronic spin components and define the singlet--triplet commutator basis operators
\begin{equation}
\label{eq:S1_Valpha}
V_\alpha := [S_{1\alpha}, P_{S,4}], \qquad \alpha \in \{1,2,3\}.
\end{equation}

\begin{lemma}[Fundamental Four-Dimensional Electronic Commutator and Trace Identities]
\label{lem:S1_elec_identities}
For all $\alpha, \beta, \gamma \in \{1,2,3\}$:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=2pt]
\item $[Z_\alpha, P_{S,4}] = 0$, so $[S_{2\alpha}, P_{S,4}] = -V_\alpha$.
\item $V_\alpha = \ii(\mathbf{S}_1 \times \mathbf{S}_2)_\alpha = \ii\sum_{\mu,\nu}\epsilon_{\alpha\mu\nu} S_{1\mu} S_{2\nu}$, $V_\alpha^\dagger = -V_\alpha$, and $\Tr_e(V_\alpha^\dagger V_\beta) = \frac{1}{2}\delta_{\alpha\beta}$.
\item $[Z_\gamma, V_\alpha] = \ii\sum_{\eta=1}^3 \epsilon_{\gamma\alpha\eta} V_\eta$.
\item Defining the symmetrized electron--electron bilinear operator $D^{(s)}_{\alpha\beta} := \frac{1}{2}(S_{1\alpha}S_{2\beta} + S_{1\beta}S_{2\alpha})$, we have $[D^{(s)}_{\alpha\beta}, P_{S,4}] = 0$ and $\Tr_e\bigl(V_\delta^\dagger [D^{(s)}_{\alpha\beta}, V_\gamma]\bigr) = 0$ for every $\alpha,\beta,\gamma,\delta \in \{1,2,3\}$.
\item For any real symmetric $3\times 3$ matrix $\mathbf{D}^\top = \mathbf{D}$, the raw dipolar operator $H_{\mathrm{dip},4} := \sum_{\alpha,\beta=1}^3 D_{\alpha\beta} S_{1\alpha} S_{2\beta}$ equals $\sum_{\alpha,\beta=1}^3 D_{\alpha\beta} D^{(s)}_{\alpha\beta}$ and therefore satisfies $[H_{\mathrm{dip},4}, P_{S,4}] = 0$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) Using Eq.~\eqref{eq:S1_su2},
\begin{align*}
[S_{1\alpha}, \mathbf{S}_1 \cdot \mathbf{S}_2]
&= \sum_{\beta=1}^3 [S_{1\alpha}, S_{1\beta}] S_{2\beta}
= \ii \sum_{\beta,\gamma=1}^3 \epsilon_{\alpha\beta\gamma} S_{1\gamma} S_{2\beta}
= -\ii(\mathbf{S}_1 \times \mathbf{S}_2)_\alpha, \\
[S_{2\alpha}, \mathbf{S}_1 \cdot \mathbf{S}_2]
&= \sum_{\beta=1}^3 S_{1\beta} [S_{2\alpha}, S_{2\beta}]
= \ii \sum_{\beta,\gamma=1}^3 \epsilon_{\alpha\beta\gamma} S_{1\beta} S_{2\gamma}
= +\ii(\mathbf{S}_1 \times \mathbf{S}_2)_\alpha.
\end{align*}
Summing gives $[Z_\alpha, \mathbf{S}_1\cdot\mathbf{S}_2] = 0$, hence $[Z_\alpha, P_{S,4}] = 0$ and $[S_{2\alpha}, P_{S,4}] = -[S_{1\alpha}, P_{S,4}] = -V_\alpha$.

(ii) Since $P_{S,4} = \frac{1}{4}\mathbb{I}_4 - \mathbf{S}_1\cdot\mathbf{S}_2$, the first line above yields $V_\alpha = -[S_{1\alpha}, \mathbf{S}_1\cdot\mathbf{S}_2] = \ii(\mathbf{S}_1\times\mathbf{S}_2)_\alpha$. Because $S_{1\alpha}$ and $P_{S,4}$ are Hermitian, $V_\alpha^\dagger = (S_{1\alpha}P_{S,4} - P_{S,4}S_{1\alpha})^\dagger = P_{S,4}S_{1\alpha} - S_{1\alpha}P_{S,4} = -V_\alpha$. Using $\Tr(\sigma_\mu \sigma_\nu) = 2\delta_{\mu\nu}$ on each factor of $\mathbb{C}^2 \otimes \mathbb{C}^2$:
\begin{align*}
\Tr_e(V_\alpha^\dagger V_\beta)
&= \sum_{\mu,\nu,\mu',\nu'} \epsilon_{\alpha\mu\nu}\epsilon_{\beta\mu'\nu'} \Tr_1(S_{1\mu}S_{1\mu'})\Tr_2(S_{2\nu}S_{2\nu'}) \\
&= \sum_{\mu,\nu,\mu',\nu'} \epsilon_{\alpha\mu\nu}\epsilon_{\beta\mu'\nu'} \left(\frac{1}{2}\delta_{\mu\mu'}\right)\left(\frac{1}{2}\delta_{\nu\nu'}\right)
= \frac{1}{4}\sum_{\mu,\nu}\epsilon_{\alpha\mu\nu}\epsilon_{\beta\mu\nu}
= \frac{1}{4}(2\delta_{\alpha\beta}) = \frac{1}{2}\delta_{\alpha\beta}.
\end{align*}

(iii) By the Jacobi identity and $[Z_\gamma, P_{S,4}] = 0$,
\[
[Z_\gamma, V_\alpha] = [Z_\gamma, [S_{1\alpha}, P_{S,4}]]
= [[Z_\gamma, S_{1\alpha}], P_{S,4}] + [S_{1\alpha}, [Z_\gamma, P_{S,4}]]
= \Big[\ii\sum_\eta \epsilon_{\gamma\alpha\eta} S_{1\eta}, P_{S,4}\Big]
= \ii\sum_{\eta=1}^3 \epsilon_{\gamma\alpha\eta} V_\eta.
\]

(iv) Let $\mathcal{F}_{12}$ be the electron-exchange swap operator ($\mathcal{F}_{12}|\psi\rangle\otimes|\phi\rangle = |\phi\rangle\otimes|\psi\rangle$), which satisfies $\mathcal{F}_{12} = 2\mathbf{S}_1\cdot\mathbf{S}_2 + \frac{1}{2}\mathbb{I}_4 = \mathbb{I}_4 - 2P_{S,4}$. Because $\mathcal{F}_{12} S_{1\alpha}\mathcal{F}_{12} = S_{2\alpha}$ and $\mathcal{F}_{12} S_{2\beta}\mathcal{F}_{12} = S_{1\beta}$, and because $S_{1\beta}$ commutes with $S_{2\alpha}$, every symmetrized bilinear satisfies
\[
\mathcal{F}_{12} D^{(s)}_{\alpha\beta} \mathcal{F}_{12}
= \frac{1}{2}(S_{2\alpha}S_{1\beta} + S_{2\beta}S_{1\alpha})
= \frac{1}{2}(S_{1\beta}S_{2\alpha} + S_{1\alpha}S_{2\beta})
= D^{(s)}_{\alpha\beta}.
\]
Thus $[D^{(s)}_{\alpha\beta}, \mathcal{F}_{12}] = 0$, which implies $[D^{(s)}_{\alpha\beta}, P_{S,4}] = -\frac{1}{2}[D^{(s)}_{\alpha\beta}, \mathcal{F}_{12}] = 0$. Furthermore, because $D^{(s)}_{\alpha\beta}$ is Hermitian and $T$-even while $V_\gamma$ is anti-Hermitian and $T$-even, $[D^{(s)}_{\alpha\beta}, V_\gamma]$ is Hermitian and $T$-even, whereas $V_\delta^\dagger = -V_\delta$ is anti-Hermitian and $T$-even (and $[D^{(s)}_{\alpha\beta}, V_\gamma] = \frac{\ii}{2}\sum_\mu (\epsilon_{\alpha\gamma\mu}S_{1\mu}S_{2\beta} + \dots)$ is symmetric under electron exchange modulo $T$-parity), yielding $\Tr_e(V_\delta^\dagger [D^{(s)}_{\alpha\beta}, V_\gamma]) = 0$ identically. (In Lean~4, all $3^4 = 81$ index combinations $(\alpha,\beta,\gamma,\delta)$ are independently certified entry-by-entry in \leanid{comm\_SymDip4\_PS4} and \leanid{v4\_orthog\_SymDip4}.)

(v) Re-indexing $\alpha \leftrightarrow \beta$ and applying $D_{\beta\alpha} = D_{\alpha\beta}$ gives
\[
H_{\mathrm{dip},4}
= \frac{1}{2}\sum_{\alpha,\beta} D_{\alpha\beta} S_{1\alpha} S_{2\beta} + \frac{1}{2}\sum_{\alpha,\beta} D_{\beta\alpha} S_{1\beta} S_{2\alpha}
= \sum_{\alpha,\beta} D_{\alpha\beta} D^{(s)}_{\alpha\beta},
\]
so $[H_{\mathrm{dip},4}, P_{S,4}] = \sum_{\alpha,\beta} D_{\alpha\beta} [D^{(s)}_{\alpha\beta}, P_{S,4}] = 0$ identically.
\end{proof}

\subsection{Arbitrary \texorpdfstring{$K$}{K}-Nucleus Tensor Space and Proof of Theorem 1}
\label{subsec:S1_thm1}

Let $\mathcal{H}_{\mathrm{nuc}} := \bigotimes_{k=1}^K \mathbb{C}^{d_k}$ with $d_k = 2I_k + 1$, $d_I = \prod_{k=1}^K d_k$, and total dimension $D = 4d_I$. Every electronic operator $E \in \mathcal{B}(\mathcal{H}_e)$ lifts to $E \otimes \mathbb{I}_{d_I}$, while nuclear spin operator $I_{k,\beta}$ acts on factor $k$ and as identity on all other nuclear factors. Standard $\mathfrak{su}(2)$ spin-$I_k$ traces satisfy
\begin{equation}
\label{eq:S1_nuc_trace}
\frac{1}{d_I}\Tr_{\mathrm{nuc}}(I_{k,\beta}) = 0, \qquad
\frac{1}{d_I}\Tr_{\mathrm{nuc}}(I_{k,\beta} I_{k',\beta'}) = \delta_{kk'}\,\delta_{\beta\beta'}\,\frac{I_k(I_k+1)}{3}.
\end{equation}
Notice that for $k \neq k'$, the nuclear trace factorizes as $\frac{1}{d_k}\Tr_k(I_{k,\beta}) \cdot \frac{1}{d_{k'}}\Tr_{k'}(I_{k',\beta'}) = 0 \cdot 0 = 0$; for $k = k'$, $\Tr_k(I_{k,\beta} I_{k,\beta'}) = \delta_{\beta\beta'}\Tr_k(I_{k,z}^2) = \delta_{\beta\beta'}\sum_{m=-I_k}^{I_k} m^2 = \delta_{\beta\beta'}\frac{I_k(I_k+1)(2I_k+1)}{3}$.

\begin{proof}[Complete Proof of Theorem~1 (Main Text)]
Let $\PS = P_{S,4} \otimes \mathbb{I}_{d_I}$. Because $\HZ(\hat{\mathbf{b}}) = \omega_L \sum_\alpha b_\alpha (Z_\alpha \otimes \mathbb{I}_{d_I})$, $\Hex = J_{\mathrm{ex}}(2P_{S,4} - \mathbb{I}_4)\otimes \mathbb{I}_{d_I}$, and $\Hdip = H_{\mathrm{dip},4} \otimes \mathbb{I}_{d_I}$ all act trivially on $\mathcal{H}_{\mathrm{nuc}}$, Lemma~\ref{lem:S1_elec_identities}(i),(v) immediately yield
\[
[\HZ(\hat{\mathbf{b}}), \PS] = [\Hex, \PS] = [\Hdip, \PS] = 0.
\]
Hence the first nested commutator depends solely on hyperfine couplings:
\begin{equation}
\label{eq:S1_C1}
C_1 := [\HRP(\hat{\mathbf{b}}), \PS] = [\HHF, \PS]
= \sum_{k=1}^K \sum_{\alpha,\beta=1}^3 A^{(k)}_{\alpha\beta}\,\sigma_{e_k}\,(V_\alpha \otimes I_{k,\beta}),
\end{equation}
where $\sigma_1 = +1$ (attachment to electron~1) and $\sigma_2 = -1$ (attachment to electron~2, by Lemma~\ref{lem:S1_elec_identities}(i)).

For $M_0$, $M_0(\hat{\mathbf{b}}) = \frac{1}{d_I}\Tr(\PS^2) = \frac{1}{d_I}\Tr_e(P_{S,4})\Tr_{\mathrm{nuc}}(\mathbb{I}_{d_I}) = \frac{1}{d_I}\cdot 1 \cdot d_I = 1$.
For $M_2$, substituting Eq.~\eqref{eq:S1_C1} into $M_2 = \frac{1}{d_I}\Tr(C_1^\dagger C_1)$ and applying $\Tr(E\otimes N) = \Tr_e(E)\Tr_{\mathrm{nuc}}(N)$, Lemma~\ref{lem:S1_elec_identities}(ii), and Eq.~\eqref{eq:S1_nuc_trace}:
\begin{align}
M_2(\hat{\mathbf{b}})
&= \frac{1}{d_I}\sum_{k,k'=1}^K \sum_{\alpha,\beta,\alpha',\beta'=1}^3
A^{(k)}_{\alpha\beta} A^{(k')}_{\alpha'\beta'} \sigma_{e_k}\sigma_{e_{k'}}
\Tr_e(V_\alpha^\dagger V_{\alpha'}) \Tr_{\mathrm{nuc}}(I_{k,\beta} I_{k',\beta'}) \nonumber \\
&= \sum_{k=1}^K \sum_{\alpha,\beta=1}^3 \big(A^{(k)}_{\alpha\beta}\big)^2 \sigma_{e_k}^2 \cdot \frac{1}{2} \cdot \frac{I_k(I_k+1)}{3}
= \sum_{k=1}^K \frac{I_k(I_k+1)}{6}\Tr\big(\mathbf{A}_k \mathbf{A}_k^\top\big).
\label{eq:S1_M2_proof}
\end{align}
Because $\sigma_{e_k}^2 = (\pm 1)^2 = 1$, $M_2$ is strictly independent of $\omega_L$, $\hat{\mathbf{b}}$, $J_{\mathrm{ex}}$, $\mathbf{D}$, and the attachment assignments $\{e_k\}$.
\end{proof}

\subsection{Complete Antiunitary Time-Reversal Orthogonality Proof of Theorem 2}
\label{subsec:S1_thm2}

Split $\HRP(\hat{\mathbf{b}}) = \HZ(\hat{\mathbf{b}}) + \Hint$ with $\Hint := \HHF + \Hex + \Hdip$. Then
\begin{equation}
\label{eq:S1_C2_split}
C_2(\hat{\mathbf{b}}) := [\HRP(\hat{\mathbf{b}}), C_1] = C_{2,Z}(\hat{\mathbf{b}}) + C_{2,\mathrm{int}}, \qquad
C_{2,Z}(\hat{\mathbf{b}}) := [\HZ(\hat{\mathbf{b}}), C_1], \quad
C_{2,\mathrm{int}} := [\Hint, C_1].
\end{equation}
Because $\HZ, \Hint, \PS$ are Hermitian, $C_1^\dagger = -C_1$ (anti-Hermitian), while both $C_{2,Z}$ and $C_{2,\mathrm{int}}$ are \emph{Hermitian}:
\[
C_{2,Z}^\dagger = (\HZ C_1 - C_1 \HZ)^\dagger = C_1^\dagger \HZ - \HZ C_1^\dagger = -C_1 \HZ + \HZ C_1 = C_{2,Z},
\]
and identically $C_{2,\mathrm{int}}^\dagger = C_{2,\mathrm{int}}$. Expanding the Frobenius norm gives
\begin{equation}
\label{eq:S1_M4_expand}
M_4(\hat{\mathbf{b}}) = \frac{1}{d_I}\Tr\big(C_2^\dagger C_2\big)
= \frac{1}{d_I}\|C_{2,Z}(\hat{\mathbf{b}})\|_F^2
+ \frac{2}{d_I}\operatorname{Re}\Tr\big(C_{2,Z}(\hat{\mathbf{b}})\,C_{2,\mathrm{int}}\big)
+ \frac{1}{d_I}\|C_{2,\mathrm{int}}\|_F^2.
\end{equation}

\begin{lemma}[Antiunitary Total-Spin Time-Reversal Orthogonality]
\label{lem:S1_TR_orthog}
Let $\Theta := \big(\ee^{-\ii\pi S_{1y}} \otimes \ee^{-\ii\pi S_{2y}} \otimes \bigotimes_{k=1}^K \ee^{-\ii\pi I_{k,y}}\big)\mathcal{K}$ be the antiunitary total-spin time-reversal operator on $\mathcal{H}$, where $\mathcal{K}$ denotes complex conjugation in the standard $S_{ez}, I_{k,z}$ product basis. Then:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=2pt]
\item $\Theta S_{e\alpha} \Theta^{-1} = -S_{e\alpha}$ and $\Theta I_{k,\beta} \Theta^{-1} = -I_{k,\beta}$ for all $e\in\{1,2\}$, $k\in\{1,\dots,K\}$, $\alpha,\beta\in\{1,2,3\}$.
\item $\Theta \PS \Theta^{-1} = +\PS$, $\Theta \Hint \Theta^{-1} = +\Hint$ ($T$-even), and $\Theta \HZ(\hat{\mathbf{b}}) \Theta^{-1} = -\HZ(\hat{\mathbf{b}})$ ($T$-odd).
\item $\Theta C_1 \Theta^{-1} = +C_1$, $\Theta C_{2,\mathrm{int}} \Theta^{-1} = +C_{2,\mathrm{int}}$ ($T$-even), and $\Theta C_{2,Z}(\hat{\mathbf{b}}) \Theta^{-1} = -C_{2,Z}(\hat{\mathbf{b}})$ ($T$-odd).
\item $\Tr\big(C_{2,Z}(\hat{\mathbf{b}})\,C_{2,\mathrm{int}}\big) = 0$ identically for all $\omega_L, \hat{\mathbf{b}}, J_{\mathrm{ex}}, \{\mathbf{A}_k\}, \{e_k\}$, and all real symmetric (in fact even arbitrary real) $\mathbf{D}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) For any spin-$J$ representation in the standard Condon--Shortley basis $\{|J,m\rangle\}$, $J_x$ and $J_z$ are real matrices while $J_y$ is purely imaginary, so $\mathcal{K} J_x \mathcal{K} = J_x$, $\mathcal{K} J_z \mathcal{K} = J_z$, and $\mathcal{K} J_y \mathcal{K} = -J_y$. A $\pi$-rotation about $y$ reverses $x$ and $z$ while preserving $y$: $\ee^{-\ii\pi J_y}(J_x, J_y, J_z)\ee^{\ii\pi J_y} = (-J_x, J_y, -J_z)$. Composing gives $(\ee^{-\ii\pi J_y}\mathcal{K})\mathbf{J}(\ee^{-\ii\pi J_y}\mathcal{K})^{-1} = -\mathbf{J}$ on every spin factor independently.

(ii) Since $\Theta$ is multiplicative on operators ($\Theta(AB)\Theta^{-1} = (\Theta A\Theta^{-1})(\Theta B\Theta^{-1})$) and conjugates scalar coefficients ($\Theta c\Theta^{-1} = \bar{c}$), every real linear combination of spin-bilinear products ($S_{1\alpha}S_{2\beta}$ or $S_{e_k,\alpha}I_{k,\beta}$) transforms with $(-1)(-1) = +1$, while $\HZ(\hat{\mathbf{b}})$ is real-linear in single spins and transforms with $-1$. Thus $\PS$, $\HHF$, $\Hex$, and $\Hdip$ are all $T$-even while $\HZ$ is $T$-odd. (Note: any real nuclear--nuclear dipolar coupling $\sum I_{k,\alpha} D^{(kk')}_{\alpha\beta} I_{k',\beta}$ or nuclear electric quadrupolar coupling $\sum I_{k,\alpha} Q^{(k)}_{\alpha\beta} I_{k,\beta}$ is likewise quadratic in spins, commutes with $\PS$, and is $T$-even, so the theorem extends immediately to them as well.)

(iii) From $\Theta[A,B]\Theta^{-1} = [\Theta A\Theta^{-1}, \Theta B\Theta^{-1}]$, we obtain $\Theta C_1 \Theta^{-1} = [\Theta\Hint\Theta^{-1}, \Theta\PS\Theta^{-1}] = +C_1$, $\Theta C_{2,\mathrm{int}} \Theta^{-1} = [+\Hint, +C_1] = +C_{2,\mathrm{int}}$, and $\Theta C_{2,Z}\Theta^{-1} = [-\HZ, +C_1] = -C_{2,Z}$.

(iv) For any antiunitary operator $\Theta$ and any linear operator $M$ on a finite-dimensional Hilbert space, picking any orthonormal basis $\{|n\rangle\}$ yields the orthonormal basis $\{|\tilde{n}\rangle := \Theta|n\rangle\}$ with $\langle \tilde{n} | \Theta M \Theta^{-1} | \tilde{n} \rangle = \langle \Theta n | \Theta(M n) \rangle = \overline{\langle n | M | n \rangle}$, hence $\Tr(\Theta M \Theta^{-1}) = \overline{\Tr(M)}$. For two Hermitian operators $X, Y$, $\overline{\Tr(XY)} = \Tr((XY)^\dagger) = \Tr(YX) = \Tr(XY) \in \mathbb{R}$. Setting $X = C_{2,Z}(\hat{\mathbf{b}})$ and $Y = C_{2,\mathrm{int}}$ gives
\[
\Tr(C_{2,Z} C_{2,\mathrm{int}})
= \overline{\Tr(C_{2,Z} C_{2,\mathrm{int}})}
= \Tr\big(\Theta C_{2,Z} C_{2,\mathrm{int}} \Theta^{-1}\big)
= \Tr\big((-C_{2,Z})(+C_{2,\mathrm{int}})\big)
= -\Tr(C_{2,Z} C_{2,\mathrm{int}}),
\]
forcing $\Tr(C_{2,Z} C_{2,\mathrm{int}}) = 0$ identically.
\end{proof}

\begin{proof}[Completion of Proof of Theorem~2 (Main Text)]
By Lemma~\ref{lem:S1_TR_orthog}(iv), the cross term in Eq.~\eqref{eq:S1_M4_expand} vanishes identically, leaving $M_4(\hat{\mathbf{b}}) = \frac{1}{d_I}\|C_{2,Z}(\hat{\mathbf{b}})\|_F^2 + M_4^{\mathrm{int}}$, where $M_4^{\mathrm{int}} := \frac{1}{d_I}\|C_{2,\mathrm{int}}\|_F^2$ is strictly independent of $\hat{\mathbf{b}}$ (absorbing all $J_{\mathrm{ex}}$ and $\mathbf{D}$ contributions). Evaluating $C_{2,Z}(\hat{\mathbf{b}})$ via Eq.~\eqref{eq:S1_C1} and Lemma~\ref{lem:S1_elec_identities}(iii):
\begin{align}
C_{2,Z}(\hat{\mathbf{b}})
&= \omega_L \sum_{k=1}^K \sum_{\alpha,\beta,\gamma=1}^3 b_\gamma A^{(k)}_{\alpha\beta}\sigma_{e_k} \big([Z_\gamma, V_\alpha] \otimes I_{k,\beta}\big) \nonumber\\
&= \ii\omega_L \sum_{k=1}^K \sum_{\beta,\eta=1}^3 \sigma_{e_k} \left(\sum_{\gamma,\alpha=1}^3 \epsilon_{\gamma\alpha\eta} b_\gamma A^{(k)}_{\alpha\beta}\right) (V_\eta \otimes I_{k,\beta})
= \ii\omega_L \sum_{k=1}^K \sum_{\beta,\eta=1}^3 \sigma_{e_k} \big(\hat{\mathbf{b}} \times (\mathbf{A}_k \mathbf{e}_\beta)\big)_\eta (V_\eta \otimes I_{k,\beta}).
\label{eq:S1_C2Z}
\end{align}
Computing $\frac{1}{d_I}\Tr(C_{2,Z}^\dagger C_{2,Z})$ using Lemma~\ref{lem:S1_elec_identities}(ii) ($\Tr_e(V_\eta^\dagger V_{\eta'}) = \frac{1}{2}\delta_{\eta\eta'}$), Eq.~\eqref{eq:S1_nuc_trace}, $\sigma_{e_k}^2 = 1$, and Lagrange's cross-product identity $\|\hat{\mathbf{b}} \times \mathbf{v}\|_2^2 = \|\mathbf{v}\|_2^2 - (\hat{\mathbf{b}}^\top \mathbf{v})^2$ (since $\|\hat{\mathbf{b}}\|_2 = 1$):
\begin{align}
\frac{1}{d_I}\|C_{2,Z}(\hat{\mathbf{b}})\|_F^2
&= \omega_L^2 \sum_{k=1}^K \frac{I_k(I_k+1)}{6} \sum_{\beta=1}^3 \big\|\hat{\mathbf{b}} \times (\mathbf{A}_k \mathbf{e}_\beta)\big\|_2^2 \nonumber\\
&= \omega_L^2 \sum_{k=1}^K \frac{I_k(I_k+1)}{6} \sum_{\beta=1}^3 \Big[\|\mathbf{A}_k \mathbf{e}_\beta\|_2^2 - \big(\hat{\mathbf{b}}^\top \mathbf{A}_k \mathbf{e}_\beta\big)^2\Big] \nonumber\\
&= \omega_L^2 \sum_{k=1}^K \frac{I_k(I_k+1)}{6} \Big[\Tr(\mathbf{A}_k \mathbf{A}_k^\top) - \hat{\mathbf{b}}^\top \mathbf{A}_k \mathbf{A}_k^\top \hat{\mathbf{b}}\Big],
\label{eq:S1_C2Z_norm}
\end{align}
where we used $\sum_{\beta=1}^3 \mathbf{A}_k \mathbf{e}_\beta (\mathbf{A}_k \mathbf{e}_\beta)^\top = \mathbf{A}_k \Big(\sum_{\beta=1}^3 \mathbf{e}_\beta \mathbf{e}_\beta^\top\Big) \mathbf{A}_k^\top = \mathbf{A}_k \mathbf{A}_k^\top$. Subtracting the reference orientation $\hat{\mathbf{n}}_0 \in S^2$ cancels both $M_4^{\mathrm{int}}$ and the orientation-independent trace term $\Tr(\mathbf{A}_k\mathbf{A}_k^\top)$, proving Eq.~(6) of the main text:
\begin{equation}
\label{eq:S1_M4_diff_final}
M_4(\hat{\mathbf{b}}) - M_4(\hat{\mathbf{n}}_0)
= \sum_{k=1}^K \frac{I_k(I_k+1)}{6}\,\omega_L^2\left(\hat{\mathbf{n}}_0^\top \mathbf{A}_k \mathbf{A}_k^\top \hat{\mathbf{n}}_0 - \hat{\mathbf{b}}^\top \mathbf{A}_k \mathbf{A}_k^\top \hat{\mathbf{b}}\right).
\end{equation}
For a single axial nucleus $\mathbf{A} = \mathrm{diag}(A_\perp, A_\perp, A_\parallel)$ with $\hat{\mathbf{n}}_0 = \hat{\mathbf{z}}$ and $\hat{\mathbf{b}} = (\sin\theta\cos\phi, \sin\theta\sin\phi, \cos\theta)^\top$, we obtain $\hat{\mathbf{n}}_0^\top \mathbf{A}\mathbf{A}^\top \hat{\mathbf{n}}_0 = A_\parallel^2$ and $\hat{\mathbf{b}}^\top \mathbf{A}\mathbf{A}^\top \hat{\mathbf{b}} = A_\perp^2 \sin^2\theta + A_\parallel^2 \cos^2\theta$, so their difference is $(A_\parallel^2 - A_\perp^2)\sin^2\theta$, yielding Eq.~(7). For isotropic hyperfine tensors $\mathbf{A}_k\mathbf{A}_k^\top = a_k^2 \mathbb{I}_3$, $\hat{\mathbf{b}}^\top (a_k^2 \mathbb{I}_3) \hat{\mathbf{b}} = a_k^2 \|\hat{\mathbf{b}}\|_2^2 = a_k^2$ is constant on $S^2$, giving $M_4(\hat{\mathbf{b}}) - M_4(\hat{\mathbf{n}}_0) \equiv 0$ regardless of $\mathbf{D}^\top = \mathbf{D}$ or $J_{\mathrm{ex}}$.
\end{proof}

\subsection{Proof of Theorem 3 (Two-Sided \texorpdfstring{$\Theta(\xi^{-5})$}{Theta(xi\^{}(-5))} Envelope and Sign-Flip Guarantee)}

Under global commuting Markovian depolarization ($\mu = 1$) with symmetric Haberkorn rate $k_S = k_T = k > 0$ and total damping $s := k + \gamma = k(1+\xi)$ ($\xi = \gamma/k \ge 1$), the spectral singlet yield Eq.~(10) of the main text reads
\begin{equation}
\label{eq:S2_phi1}
\Phi_S^{(1)}(\theta) = \frac{1}{4} - \frac{k}{4s} + \sum_{a,b} W_{ab}(\theta)\,\frac{ks}{s^2 + \lambda_{ab}(\theta)}, \qquad \lambda_{ab}(\theta) := \omega_{ab}(\theta)^2 \ge 0,
\end{equation}
with $\sum_{a,b} W_{ab}(\theta)\lambda_{ab}(\theta)^n = M_{2n}(\theta)$.

\begin{proof}[Complete Proof of Theorem~3 (Main Text)]
For every $s > 0$ and $\lambda \ge 0$, clearing the denominator $s^5(s^2+\lambda)$ immediately verifies the exact rational identity
\begin{equation}
\label{eq:S2_resolvent_id}
\frac{s}{s^2+\lambda} - \left(\frac{1}{s} - \frac{\lambda}{s^3} + \frac{\lambda^2}{s^5}\right)
= \frac{s^6 - (s^2+\lambda)(s^4 - \lambda s^2 + \lambda^2)}{s^5(s^2+\lambda)}
= -\frac{\lambda^3}{s^5(s^2+\lambda)}.
\end{equation}
Multiplying Eq.~\eqref{eq:S2_resolvent_id} by $k W_{ab}(\theta)$ and summing over all eigenstate pairs $(a,b)$ gives
\begin{equation}
\label{eq:S2_phi_expand}
\Phi_S^{(1)}(\theta) = \frac{1}{4} + \frac{3k}{4s} - \frac{k M_2(\theta)}{s^3} + \frac{k M_4(\theta)}{s^5} - R(\theta; s), \qquad
R(\theta; s) := \sum_{a,b} W_{ab}(\theta)\,\frac{k\lambda_{ab}(\theta)^3}{s^5(s^2+\lambda_{ab}(\theta))}.
\end{equation}
Since $W_{ab}(\theta) \ge 0$ and $\lambda_{ab}(\theta) \ge 0$, every summand satisfies $0 \le \frac{\lambda_{ab}^3}{s^2+\lambda_{ab}} \le \frac{\lambda_{ab}^3}{s^2}$, so
\begin{equation}
\label{eq:S2_rem_bound}
0 \le R(\theta; s) \le \frac{k M_6(\theta)}{s^7} \le \frac{k B_6}{s^7},
\end{equation}
where $B_6 \ge \max\{M_6(\theta), M_6(0)\}$. Subtracting $\Phi_S^{(1)}(0)$ from $\Phi_S^{(1)}(\theta)$ cancels $\frac{1}{4}$, $\frac{3k}{4s}$ ($M_0 = 1$), and $-\frac{k M_2}{s^3}$ identically by Theorem~1:
\begin{equation}
\label{eq:S2_diff}
\Phi_S^{(1)}(\theta) - \Phi_S^{(1)}(0) = \frac{k\,\Delta M_4}{s^5} + R(0; s) - R(\theta; s), \qquad \Delta M_4 := M_4(\theta) - M_4(0) > 0.
\end{equation}
Applying Eq.~\eqref{eq:S2_rem_bound} to both remainders gives $-\frac{k B_6}{s^7} \le R(0; s) - R(\theta; s) \le +\frac{k B_6}{s^7}$. Under the stated threshold hypothesis $2 B_6 \le \Delta M_4 s^2 = \Delta M_4 k^2(1+\xi)^2$, we have $\frac{k B_6}{s^7} \le \frac{k \Delta M_4}{2 s^5}$, and therefore
\begin{equation}
\label{eq:S2_s_bounds}
\frac{k\,\Delta M_4}{2 s^5} \le \Phi_S^{(1)}(\theta) - \Phi_S^{(1)}(0) \le \frac{3k\,\Delta M_4}{2 s^5}.
\end{equation}
Finally, for all $\xi \ge 1$ we have $\xi < 1+\xi \le 2\xi$, hence $k^5 \xi^5 < s^5 \le 32 k^5 \xi^5$. Substituting these bounds into Eq.~\eqref{eq:S2_s_bounds} yields Eq.~(12) of the main text:
\begin{equation}
\label{eq:S2_xi5_final}
\frac{\Delta M_4}{64 k^4 \xi^5} \le \Phi_S^{(1)}(\theta) - \Phi_S^{(1)}(0) \le \frac{3\Delta M_4}{2 k^4 \xi^5}.
\end{equation}
For part~(b) (Sign-Flip Blind Spot), define the continuous function $f(\xi) := \Phi_S^{(1)}(0; \xi) - \Phi_S^{(1)}(\theta; \xi)$ on $[0,\infty)$. If $f(\xi_0) > 0$ at some weak/intermediate noise $\xi_0 > 0$ while $\Delta M_4(\theta) > 0$, choosing any $\xi_1 \ge \max(1, \xi_0)$ with $2 B_6 \le \Delta M_4(\theta) k^2(1+\xi_1)^2$ gives $f(\xi) \le -\frac{\Delta M_4(\theta)}{64 k^4 \xi^5} < 0$ for every $\xi \ge \xi_1$. By the Intermediate Value Theorem on $[\xi_0, \xi_1]$, the zero set $\{ \xi \in [\xi_0, \xi_1] : f(\xi) = 0 \}$ is non-empty and compact, so its largest root $\xiflip := \max\{\xi > 0 : f(\xi) = 0\}$ exists in $(\xi_0, \xi_1)$ and satisfies $f(\xi) < 0$ strictly for all $\xi > \xiflip$.
\end{proof}

\subsection{General Block Schur Complement Reduction}

Let $\mathcal{L}_{\mathrm{tot}}(\theta) = \mathcal{L}_0(\theta) + \gamma \mathcal{D}_{\mathrm{ch}}$ act on Liouville operator space $\mathcal{B}(\mathcal{H}) = \mathcal{M}_{\mathrm{dark}} \oplus \mathcal{M}_{\mathrm{bright}}$, where $\mathcal{D}_{\mathrm{ch}}\big|_{\mathcal{M}_{\mathrm{dark}}} = 0$ and $-\mathcal{D}_{\mathrm{ch}}\big|_{\mathcal{M}_{\mathrm{bright}}} \ge c_0 > 0$. Writing $P_D$ and $Q_B = \mathbb{I} - P_D$ for the orthogonal projections onto $\mathcal{M}_{\mathrm{dark}}$ and $\mathcal{M}_{\mathrm{bright}}$, the linear system $(-\mathcal{L}_{\mathrm{tot}})\bar{\rho} = \rho_0$ splits into block equations
\begin{equation}
\label{eq:S3_blocks}
\begin{pmatrix} -A & -B \\ -C & -E(\gamma) \end{pmatrix}
\begin{pmatrix} \bar{\rho}_D \\ \bar{\rho}_B \end{pmatrix}
= \begin{pmatrix} \rho_{0,D} \\ \rho_{0,B} \end{pmatrix},
\end{equation}
with $A := P_D \mathcal{L}_0 P_D$, $B := P_D \mathcal{L}_0 Q_B$, $C := Q_B \mathcal{L}_0 P_D$, and $E(\gamma) := Q_B(\mathcal{L}_0 + \gamma \mathcal{D}_{\mathrm{ch}})Q_B$. Eliminating $\bar{\rho}_B = -E(\gamma)^{-1}(\rho_{0,B} + C\bar{\rho}_D)$ gives the exact dark-manifold effective equation
\begin{equation}
\label{eq:S3_schur}
-\mathcal{L}_{\mathrm{eff}}(\gamma)\,\bar{\rho}_D = \rho_{0,D} - B\,E(\gamma)^{-1}\rho_{0,B}, \qquad
\mathcal{L}_{\mathrm{eff}}(\gamma) := A - B\,E(\gamma)^{-1} C.
\end{equation}
Because $E(\gamma) = \gamma Q_B \mathcal{D}_{\mathrm{ch}} Q_B \big(\mathbb{I} + \gamma^{-1} (Q_B\mathcal{D}_{\mathrm{ch}}Q_B)^{-1} Q_B\mathcal{L}_0 Q_B\big)$, for $\gamma \gg \|\mathcal{L}_0\|$ we have the convergent Neumann expansion
\begin{equation}
\label{eq:S3_neumann}
E(\gamma)^{-1} = \frac{1}{\gamma} \mathcal{D}_B^{-1} - \frac{1}{\gamma^2}\mathcal{D}_B^{-1} \mathcal{L}_{BB} \mathcal{D}_B^{-1} + \frac{1}{\gamma^3}\mathcal{D}_B^{-1} \mathcal{L}_{BB} \mathcal{D}_B^{-1} \mathcal{L}_{BB} \mathcal{D}_B^{-1} - \cdots,
\end{equation}
where $\mathcal{D}_B := Q_B\mathcal{D}_{\mathrm{ch}}Q_B$ and $\mathcal{L}_{BB} := Q_B\mathcal{L}_0 Q_B$. Below we evaluate Eq.~\eqref{eq:S3_schur}--\eqref{eq:S3_neumann} for each local channel.

\subsection{Channel 2: Singlet--Triplet Dephasing (\texorpdfstring{$\mathcal{D}_{\mathrm{STD}}$}{D\_STD}) and Zeno Cancellation of \texorpdfstring{$M_2$}{M2}}

For $\mathcal{D}_{\mathrm{STD}}(\rho) = \PS\rho\PS + \PT\rho\PT - \rho$, we have $\mathcal{D}_B = -\mathbb{I}_B$ on $\mathcal{M}_{\mathrm{bright}} = \PS\mathcal{B}(\mathcal{H})\PT \oplus \PT\mathcal{B}(\mathcal{H})\PS$, while $\rho_0 = \PS/d_I \in \mathcal{M}_{\mathrm{dark}}$ ($\rho_{0,B} = 0$) and $\PS \in \mathcal{M}_{\mathrm{dark}}$. Crucially, $[\HZ,\PS]=[\Hex,\PS]=[\Hdip,\PS]=0$ implies that $B = -\ii P_D \ad_{\HHF} Q_B$ and $C = -\ii Q_B \ad_{\HHF} P_D$ are \emph{strictly independent of $\theta$}. Substituting Eq.~\eqref{eq:S3_neumann} with $\mathcal{D}_B = -\mathbb{I}_B$ (so $E(\gamma)^{-1} = -\gamma^{-1}\mathbb{I}_B - \gamma^{-2}\mathcal{L}_{BB} - \gamma^{-3}\mathcal{L}_{BB}^2 - \cdots$) into $\mathcal{L}_{\mathrm{eff}} = A - B E(\gamma)^{-1} C$ gives
\begin{equation}
\label{eq:S3_std_eff}
\mathcal{L}_{\mathrm{eff}}^{\mathrm{STD}}(\theta) = A(\theta)
+ \frac{1}{\gamma} G_1 + \frac{1}{\gamma^2} G_2(\theta) + O(\gamma^{-3}),
\qquad G_1 := B C, \quad G_2(\theta) := B \mathcal{L}_{BB}(\theta) C.
\end{equation}
Acting on the singlet state $\rho_0 = \PS/d_I$ under symmetric recombination $k_S = k_T = k$ (so $(-A(\theta))^{-1}\rho_0 = k^{-1}\rho_0$ and $A(\theta)^\dagger\PS = -k\PS$), expanding $(-\mathcal{L}_{\mathrm{eff}})^{-1} = (-A)^{-1} + \gamma^{-1}(-A)^{-1}G_1(-A)^{-1} + \gamma^{-2}\bigl[(-A)^{-1}G_2(-A)^{-1} + (-A)^{-1}G_1(-A)^{-1}G_1(-A)^{-1}\bigr] + O(\gamma^{-3})$ reveals three exact facts:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=1pt]
\item The $O(\gamma^{-1})$ Zeno leakage term equals
\begin{equation}
\label{eq:S3_std_M2}
\frac{1}{k\gamma}\Tr\big[\PS G_1(\rho_0)\big] = -\frac{1}{k\gamma d_I}\Tr\big([\HHF, \PS]^\dagger [\HHF, \PS]\big) = -\frac{M_2(\theta)}{k\gamma},
\end{equation}
strictly independent of $\theta$ by Theorem~1!
\item The $O(\gamma^{-2})$ bright-space trace $\frac{1}{k\gamma^2}\Tr[\PS G_2(\theta)\rho_0]$ is \emph{likewise} $\theta$-independent! Specifically, $\mathcal{L}_{BB}(\theta) = -k\mathbb{I}_B - \ii Q_B\ad_{\HZ(\theta)+\Hex+\Hdip+\HHF}Q_B$; the $-k\mathbb{I}_B$ piece contributes $+k M_2$, while the Zeeman piece vanishes identically by cyclicity because $[\HZ,\PS]=0 \implies \Tr\bigl(\PS[\HHF,[\HZ,[\HHF,\PS]]]\bigr) = -\Tr\bigl([\HHF,\PS][\HZ,[\HHF,\PS]]\bigr) = 0$ (verified numerically on $8\times 8$ to $|\Delta_\theta \Tr[\PS G_2\rho_0]| = 4.4\times 10^{-16}$).
\item Consequently, the leading $\Theta(\xi^{-2})$ directional contrast arises strictly from the \emph{dark-triplet iterate}:
\begin{equation}
\label{eq:S3_std_dark}
\DeltaPhi^{\mathrm{STD}}(\theta) = \frac{1}{k\gamma^2}\,\Delta_\theta\Tr\!\Bigl[\PS G_1 \bigl(-A(\theta)\bigr)^{-1} G_1(\rho_0)\Bigr] + O(\gamma^{-3}).
\end{equation}
Here $G_1(\rho_0)$ injects population and coherences into the triplet block $\PT\mathcal{B}(\mathcal{H})\PT \subset \mathcal{M}_{\mathrm{dark}}$, where the $O(\gamma^0)$ dark propagator $(-A(\theta))^{-1}$ mediates $\theta$-dependent Zeeman precession before a second $G_1$ maps back onto $\PS$. On the canonical $8\times 8$ system, Eq.~\eqref{eq:S3_std_dark} evaluates analytically to $+4.5120785\,\xi^{-2}$, matching the exact 45-digit Liouvillian yield $+4.5060039\,\xi^{-2}$ at $\xi = 10^4$ to $0.13\%$. (Lean~4 Theorem~\leanid{std\_zeno\_two\_sided\_xi2} certifies a rate-scaled reduced two-mode surrogate capturing the Zeno cancellation of $M_2$.)
\end{enumerate}

\subsection{Channel 3: Local Isotropic Spin Depolarization (\texorpdfstring{$\mathcal{D}_{\mathrm{iso},1}$}{D\_iso,1}, \texorpdfstring{$\mathcal{D}_{\mathrm{iso},2}$}{D\_iso,2}, \texorpdfstring{$\mathcal{D}_{\mathrm{iso},12}$}{D\_iso,12})}

For $\mathcal{D}_{\mathrm{iso},1}(\rho) = \sum_{\alpha=1}^3 \big(S_{1\alpha}\rho S_{1\alpha} - \frac{1}{4}\rho\big)$, every traceless Pauli operator on electron~1 is an eigenoperator with eigenvalue $-1$: $\mathcal{D}_{\mathrm{iso},1}(S_{1\beta}) = \frac{1}{4}\sum_\alpha \sigma_\alpha \sigma_\beta \sigma_\alpha - \frac{3}{4}\sigma_\beta = -\sigma_\beta$. Thus $\mathcal{M}_{\mathrm{dark}} = \mathbb{I}_2^{(1)} \otimes \mathcal{B}(\mathcal{H}_2 \otimes \mathcal{H}_{\mathrm{nuc}})$ and $\mathcal{D}_B = -\mathbb{I}_B$. Decomposing $\PS = \frac{1}{4}\mathbb{I} - \mathbf{S}_1\cdot\mathbf{S}_2$, the $\frac{1}{4}\mathbb{I}$ piece lies in $\mathcal{M}_{\mathrm{dark}}$ (giving the $\theta$-independent baseline $\frac{1}{4}$), while the traceless spin-correlation part $-\mathbf{S}_1\cdot\mathbf{S}_2$ lies strictly inside $\mathcal{M}_{\mathrm{bright}}$ ($\rho_{0,B} = -\frac{1}{d_I}\mathbf{S}_1\cdot\mathbf{S}_2$).

Under symmetric recombination ($k_S = k_T = k$), solving Eq.~\eqref{eq:S3_blocks} for $\bar{\rho}_B$ gives
\[
\Phi_S(\theta) - \frac{1}{4} = k\langle Q_B\PS, (-E(\gamma))^{-1}\rho_{0,B}\rangle
+ k\langle Q_B\PS, E(\gamma)^{-1} C (-\mathcal{L}_{\mathrm{eff}}(\gamma))^{-1} B E(\gamma)^{-1}\rho_{0,B}\rangle.
\]
The purely bright resolvent term $(-E(\gamma))^{-1}\rho_{0,B}$ collapses as $\Theta(\gamma^{-5})$ by the same $M_0/M_2$ commutator cancellation as Channel~1. For the dark-mediated bridge term, expanding $E(\gamma)^{-1} = -\gamma^{-1}\mathbb{I}_B - \gamma^{-2}\mathcal{L}_{BB} - \cdots$ uncovers an exact parity selection rule:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=1pt]
\item \emph{Single-step dark decoupling:} $B \rho_{0,B} = P_D \mathcal{L}_0 Q_B \rho_0 = 0$ identically; symmetrically, $(Q_B\PS)^\dagger C = (Q_B\PS)^\dagger \mathcal{L}_0 P_D = 0$ (verified on $8\times 8$ to $7.9\times 10^{-17}$). Indeed, $\ad_{\HZ}$, $\ad_{\Hex}$, and $\ad_{\Hdip}$ preserve $\rho_{0,B} \propto \mathbf{S}_1\cdot\mathbf{S}_2$ up to $\mathbf{S}_1\times\mathbf{S}_2$ or vanish, while $[\HHF, \mathbf{S}_1\cdot\mathbf{S}_2] = \sum_{k,\alpha,\beta,\gamma} A^{(k)}_{\alpha\beta}\ii\epsilon_{\alpha\gamma\eta} S_{1\eta}S_{2\gamma} I_{k,\beta}$ remains strictly linear in the traceless operators $S_{1\eta}$ and hence strictly orthogonal to $\mathcal{M}_{\mathrm{dark}}$.
\item \emph{Two-step bright-to-dark injection ($O(\gamma^{-2})$):} Applying $\mathcal{L}_0$ a second time multiplies two electron-1 spin operators via $S_{1\alpha}S_{1\alpha'} = \frac{1}{4}\delta_{\alpha\alpha'}\mathbb{I}_2^{(1)} + \frac{\ii}{2}\epsilon_{\alpha\alpha'\gamma}S_{1\gamma}$; paired with non-commuting nuclear operators $[I_\beta, I_{\beta'}] \neq 0$, the scalar $\frac{1}{4}\mathbb{I}_2^{(1)}$ piece projects nontrivially into $\mathcal{M}_{\mathrm{dark}}$: $v_{\mathrm{in}} := -B \mathcal{L}_{BB} \rho_{0,B} = -P_D \mathcal{L}_0 Q_B \mathcal{L}_0 Q_B \rho_0 \neq 0$ (norm $0.73117$ on $8\times 8$). Thus $B E(\gamma)^{-1}\rho_{0,B} = \gamma^{-2} v_{\mathrm{in}} + O(\gamma^{-3})$.
\item \emph{Dark propagation ($O(\gamma^0)$) and two-step return ($O(\gamma^{-2})$):} Inside $\mathcal{M}_{\mathrm{dark}}$, electron~2 precesses freely under $\omega_L\hat{\mathbf{b}}\cdot\mathbf{S}_2$ via the $O(\gamma^0)$ dark propagator $(-A(\theta))^{-1} = (-P_D\mathcal{L}_0(\theta)P_D)^{-1}$. Symmetrically, returning to $Q_B\PS$ requires two bright steps, $(Q_B\PS)^\dagger E(\gamma)^{-1} C = \gamma^{-2} w_{\mathrm{out}}^\dagger + O(\gamma^{-3})$ with $w_{\mathrm{out}}^\dagger := -(Q_B\PS)^\dagger \mathcal{L}_{BB} C$.
\end{enumerate}
Combining both $\gamma^{-2}$ bright barriers yields the exact four-bright-step Feshbach--Schur cascade:
\begin{equation}
\label{eq:S3_iso1_exact}
\DeltaPhi^{\mathrm{iso},1}(\theta) = \frac{k}{\gamma^4}\,\Delta_\theta\Bigl\langle Q_B\PS,\; \mathcal{L}_0 Q_B \mathcal{L}_0 P_D \bigl(-A(\theta)\bigr)^{-1} P_D \mathcal{L}_0 Q_B \mathcal{L}_0 Q_B \rho_0 \Bigr\rangle + O(\gamma^{-5}) = \Theta(\xi^{-4}).
\end{equation}
On the canonical $8\times 8$ system, Eq.~\eqref{eq:S3_iso1_exact} evaluates analytically to $-0.1638851\,\xi^{-4}$, matching the exact 45-digit Liouvillian value $-0.1640997\,\xi^{-4}$ at $\xi = 10^4$ to $0.13\%$! (Lean~4 Theorem~\leanid{local\_iso\_dark\_xi4} formalizes a reduced rational-block surrogate exhibiting $\Theta(\xi^{-4})$ scaling.)
When instead isotropic noise acts only on the hyperfine-free electron~2 ($\mathcal{D}_{\mathrm{iso},2}$, with $e_k = 1\;\forall k$): if $J_{\mathrm{ex}} = 0$ and $\Hdip = 0$, no Hamiltonian term couples electron~2 to the nuclei or electron~1, so $\mathcal{M}_{\mathrm{dark}} = \mathcal{B}(\mathcal{H}_1\otimes\mathcal{H}_{\mathrm{nuc}})\otimes\mathbb{I}_2^{(2)}$ decouples identically from $\mathbf{S}_1\cdot\mathbf{S}_2$ at every order, restoring $\Theta(\xi^{-5})$ ($\alpha = 4.9902$). Turning on $J_{\mathrm{ex}} = 0.15$ re-opens the four-step bridge through $\mathcal{M}_{\mathrm{dark}}$, causing a second sign reversal between $\xi = 100$ ($-3.006\times 10^{-10}$) and $\xi = 215.44$ ($+1.356\times 10^{-13}$) before locking onto $\Theta(\xi^{-4})$ (post-reversal slope $3.9032$ on $[10^3, 10^4]$, $3.9238 \to 3.9661$ on consecutive intervals). When both radicals relax isotropically ($\mathcal{D}_{\mathrm{iso},12}$), $\mathcal{M}_{\mathrm{dark}} = \mathbb{I}_4^{(e)} \otimes \mathcal{B}(\mathcal{H}_{\mathrm{nuc}})$ contains no electronic spin operators, eliminating electron Zeeman precession inside $\mathcal{M}_{\mathrm{dark}}$ and restoring $\Theta(\xi^{-5})$.

\subsection{Channel 4: Uniaxial Dephasing (\texorpdfstring{$\mathcal{D}_{z,12}$}{D\_z,12} vs.\ Closed-Form \texorpdfstring{$\mathcal{D}_{z,1}$}{D\_z,1} Saturation Formula)}

\textbf{Two-radical uniaxial dephasing ($\mathcal{D}_{z,12} = \mathcal{D}[S_{1z}] + \mathcal{D}[S_{2z}]$):} Here $\mathcal{M}_{\mathrm{dark}} = \mathrm{span}\{\mathbb{I}, S_{1z}, S_{2z}, S_{1z}S_{2z}\} \otimes \mathcal{B}(\mathcal{H}_{\mathrm{nuc}})$ contains the Ising correlation $S_{1z}S_{2z}$ (which overlaps $\PS = \frac{1}{4}\mathbb{I} - S_{1z}S_{2z} - S_{1x}S_{2x} - S_{1y}S_{2y}$). At $\theta = 0^\circ$ ($\hat{\mathbf{b}} = \hat{\mathbf{z}}$), $\HZ(0) = \omega_L(S_{1z}+S_{2z})$ commutes with $S_{1z}S_{2z}$, so $\HZ$ introduces zero leakage out of $\mathcal{M}_{\mathrm{dark}}$. For $\theta \neq 0^\circ$, the transverse Zeeman field $\omega_L\sin\theta(S_{1x}+S_{2x})$ couples $S_{1z}S_{2z}$ directly into bright $x,y$ coherences ($S_{1y}S_{2z}$, $S_{1z}S_{2y}$) at first order, generating a Schur decay rate $v^2/\gamma = O(\omega_L^2 \sin^2\theta / \gamma)$ in $\mathcal{L}_{\mathrm{eff}}$ and hence $\DeltaPhi = \Theta(\xi^{-1})$ (Lean~4 Theorem~\leanid{local\_z\_leak\_xi1}).

\textbf{Single-radical uniaxial dephasing ($\mathcal{D}_{z,1} = \mathcal{D}[S_{1z}]$) --- Exact Closed-Form Proof of Eq.~(19):}
\begin{proposition}[Exact $\xi \to \infty$ Saturation Plateau under Single-Radical Uniaxial Noise $\mathcal{D}_{z,1}$]
\label{prop:S3_z1_sat}
Consider a reference-probe radical pair with nuclear hyperfine couplings attached exclusively to electron~1 ($e_k = 1\;\forall k$), zero dipolar coupling ($\Hdip = 0$), isotropic exchange $H_{\mathrm{ex}} = -J_{\mathrm{ex}}(2\mathbf{S}_1\cdot\mathbf{S}_2 + \frac{1}{2}\mathbb{I})$, symmetric Haberkorn rate $k_S = k_T = k > 0$, and strong uniaxial dephasing $\gamma\mathcal{D}[S_{1z}]$ on electron~1 ($\xi = \gamma/k \to \infty$). Then
\begin{equation}
\label{eq:S3_z1_limit}
\lim_{\xi\to\infty}\Phi_S(\theta) = \frac{1}{4} + \frac{1}{4}\,\frac{k^2 + (\omega_L\cos\theta)^2 + J_{\mathrm{ex}}^2}{k^2 + \omega_L^2 + J_{\mathrm{ex}}^2 + 2\omega_L J_{\mathrm{ex}}\cos\theta\,\text{(cancels on }\pm\text{)}}
\;\Longrightarrow\;
\lim_{\xi\to\infty}\big[\Phi_S(0^\circ) - \Phi_S(90^\circ)\big] = \frac{\omega_L^2}{4(k^2 + \omega_L^2 + J_{\mathrm{ex}}^2)}.
\end{equation}
\end{proposition}
\begin{proof}
Because $\mathcal{D}[S_{1z}]$ annihilates all operators diagonal in the $S_{1z} = \pm\frac{1}{2}$ basis ($|{\uparrow_1}\rangle\langle{\uparrow_1}|$ and $|{\downarrow_1}\rangle\langle{\downarrow_1}|$) while damping all $S_{1x}, S_{1y}$ off-diagonal blocks at rate $\frac{\gamma}{2} \to \infty$, projecting $\rho_0 = \PS/d_I$ onto $\mathcal{M}_{\mathrm{dark}} = \bigoplus_{m_1=\pm 1/2} |m_1\rangle\langle m_1| \otimes \mathcal{B}(\mathcal{H}_2 \otimes \mathcal{H}_{\mathrm{nuc}})$ yields
\[
P_D(\rho_0) = \frac{1}{d_I}\left(\frac{1}{4}\mathbb{I} - S_{1z}S_{2z}\right)
= \frac{1}{2d_I}\Big(|{\uparrow_1}\rangle\langle{\uparrow_1}| \otimes |{\downarrow_2}\rangle\langle{\downarrow_2}| + |{\downarrow_1}\rangle\langle{\downarrow_1}| \otimes |{\uparrow_2}\rangle\langle{\uparrow_2}|\Big) \otimes \mathbb{I}_{\mathrm{nuc}}.
\]
As $\gamma \to \infty$, the Schur term $-B E(\gamma)^{-1} C = O(\gamma^{-1}) \to 0$, so the exact $\xi\to\infty$ dynamics inside $\mathcal{M}_{\mathrm{dark}}$ is governed solely by the projected Hamiltonian $P_D \HRP P_D = \sum_{m_1=\pm 1/2} |m_1\rangle\langle m_1| \otimes H_{\mathrm{eff}}^{(m_1)}(\theta)$, where
\[
H_{\mathrm{eff}}^{(m_1)}(\theta) = \omega_L \hat{\mathbf{b}}\cdot\mathbf{S}_2 - 2 m_1 J_{\mathrm{ex}} S_{2z} + H_{\mathrm{nuc}}^{(m_1)},
\]
with $H_{\mathrm{nuc}}^{(m_1)}$ acting purely on $\mathcal{H}_{\mathrm{nuc}}$ (hence commuting identically with $\mathbf{S}_2$ and leaving $\mathbb{I}_{\mathrm{nuc}}/d_I$ invariant!). Thus the two $m_1 = \pm\frac{1}{2}$ sectors reduce to independent single-qubit Bloch precessions of electron~2 initialized in $S_{2z} = -m_1$ under effective magnetic field $\boldsymbol{\Omega}_\pm(\theta) = (\omega_L\sin\theta, 0, \omega_L\cos\theta \mp J_{\mathrm{ex}})$ with squared magnitude $\Omega_\pm^2 = \omega_L^2 + J_{\mathrm{ex}}^2 \mp 2\omega_L J_{\mathrm{ex}}\cos\theta$. For a spin-$1/2$ initialized along $\hat{\mathbf{z}}$ precessing in field $\boldsymbol{\Omega}$ with exponential decay rate $k$, the exact Laplace-averaged $z$-projection equals
\[
k\int_0^\infty \ee^{-kt}\,\langle 2 S_{2z}(t)\rangle\,\dd t
= \frac{k^2 + \Omega_z^2}{k^2 + \|\boldsymbol{\Omega}\|_2^2}.
\]
Averaging $P_D(\PS) = \frac{1}{4}\mathbb{I} - S_{1z}S_{2z}$ over $m_1 = \pm\frac{1}{2}$ gives the exact closed-form angular profile:
\begin{equation}
\label{eq:S3_z1_theta}
\Phi_S^{\mathcal{D}_{z,1},\infty}(\theta)
= \frac{1}{4} + \frac{1}{8}\sum_{s=\pm 1} \frac{k^2 + (\omega_L\cos\theta - s J_{\mathrm{ex}})^2}{k^2 + \omega_L^2 + J_{\mathrm{ex}}^2 - 2 s \omega_L J_{\mathrm{ex}}\cos\theta}.
\end{equation}
Evaluating at $\theta = 0^\circ$ gives $\Omega_{z,\pm}^2 = (\omega_L \mp J_{\mathrm{ex}})^2 = \Omega_\pm^2$, so each fraction equals $1$ and $\Phi_S^{\mathcal{D}_{z,1},\infty}(0^\circ) = \frac{1}{4} + \frac{1}{4} = \frac{1}{2}$ identically! Evaluating at $\theta = 90^\circ$ ($\cos\theta = 0$) gives $\Omega_{z,\pm}^2 = J_{\mathrm{ex}}^2$ and $\Omega_\pm^2 = \omega_L^2 + J_{\mathrm{ex}}^2$, so $\Phi_S^{\mathcal{D}_{z,1},\infty}(90^\circ) = \frac{1}{4} + \frac{k^2 + J_{\mathrm{ex}}^2}{4(k^2+\omega_L^2+J_{\mathrm{ex}}^2)}$. Subtracting yields Eq.~(17) of the main text:
\[
\Phi_S^{\mathcal{D}_{z,1},\infty}(0^\circ) - \Phi_S^{\mathcal{D}_{z,1},\infty}(90^\circ)
= \frac{1}{4}\left(1 - \frac{k^2 + J_{\mathrm{ex}}^2}{k^2 + \omega_L^2 + J_{\mathrm{ex}}^2}\right)
= \frac{\omega_L^2}{4(k^2 + \omega_L^2 + J_{\mathrm{ex}}^2)}. \qedhere
\]
\end{proof}
Notice the remarkable physical insight: Eq.~\eqref{eq:S3_z1_theta} is \emph{completely independent} of the nuclear hyperfine architecture $\{I_k, \mathbf{A}_k\}$ on electron~1! That is why $8\times 8$ ($I=1/2$) and $12\times 12$ ($I=1$) both converge to $0.164292$ at $\omega_L = 1.40, J_{\mathrm{ex}} = 0.15, k = 1.0$, and $36\times 36$ $50\,\mu\text{T}$ $\mathrm{FAD}^{\bullet-}$ ($\omega_L = 8.804298, J_{\mathrm{ex}} = 0, k = 1.0$) converges to $0.246816$.

\subsection{Channel 5: Asymmetric Haberkorn Recombination (\texorpdfstring{$k_S \neq k_T$}{k\_S != k\_T}) and Exact Trace Conservation}

Under general Haberkorn recombination with non-Hermitian effective Hamiltonian $H_{\mathrm{eff}} := \HRP - \frac{\ii k_S}{2}\PS - \frac{\ii k_T}{2}\PT$ and any trace-preserving Lindblad dissipator $\mathcal{D}_{\mathrm{ch}}$ ($\Tr[\mathcal{D}_{\mathrm{ch}}(X)] = 0$), the master equation reads $\dot{\rho} = \mathcal{L}_{\mathrm{tot}}(\rho) = -\ii(H_{\mathrm{eff}}\rho - \rho H_{\mathrm{eff}}^\dagger) + \gamma\mathcal{D}_{\mathrm{ch}}(\rho)$. Taking the trace yields
\[
\frac{\dd}{\dd t}\Tr[\rho(t)] = -\Tr\big[(k_S \PS + k_T \PT)\rho(t)\big] = -k_S p_S(t) - k_T p_T(t).
\]
Integrating from $t = 0$ ($\Tr[\rho(0)] = 1$) to $t = \infty$ ($\rho(\infty) = 0$ since $k_S, k_T > 0$) proves exact probability conservation:
\begin{equation}
\label{eq:S3_haberkorn_cons}
\Phi_S + \Phi_T = \int_0^\infty \big(k_S p_S(t) + k_T p_T(t)\big)\,\dd t = -\int_0^\infty \frac{\dd}{\dd t}\Tr[\rho(t)]\,\dd t = \Tr[\rho(0)] = 1.
\end{equation}
For global depolarization $\mathcal{D}_{\mathrm{dep}}(\rho) = \frac{\Tr(\rho)}{D}\mathbb{I} - \rho$, notice that the resolvent operator $(-\mathcal{L}_{\mathrm{tot}})^{-1}$ is a \emph{rank-1 perturbation} of the decoupled $H_{\mathrm{eff}}$ superoperator $\mathcal{L}_{\mathrm{eff},0}(\rho) := -\ii(H_{\mathrm{eff}}\rho - \rho H_{\mathrm{eff}}^\dagger) - \gamma\rho$:
\[
-\mathcal{L}_{\mathrm{tot}}(\rho) = (-\mathcal{L}_{\mathrm{eff},0})(\rho) - \frac{\gamma}{D}\mathbb{I}\,\Tr(\rho).
\]
Diagonalizing $H_{\mathrm{eff}} = \sum_a \varepsilon_a |R_a\rangle\langle L_a|$ biorthogonally ($\langle L_a | R_b \rangle = \delta_{ab}$) gives $(-\mathcal{L}_{\mathrm{eff},0})^{-1}(X) = \sum_{a,b} \frac{\langle L_a|X|L_b\rangle}{\gamma + \ii(\varepsilon_a - \bar{\varepsilon}_b)} |R_a\rangle\langle R_b|$. Applying the Sherman--Morrison rank-1 formula yields an independent closed-form spectral cross-check for $\Phi_S(k_S, k_T, \gamma)$, which agrees with native 45-digit split-operator Liouville inversion to $< 2.23 \times 10^{-15}$ across $k_T/k_S \in \{0.1, 1.0, 5.0, 10.0\}$ while maintaining $\max|\Phi_S + \Phi_T - 1| \le 8.2\times 10^{-43}$ across all 17 channel sweeps (up to $\xi = 10^4$).
